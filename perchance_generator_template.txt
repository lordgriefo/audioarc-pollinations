  Whip pan transition snapping from the character's reaction to the surrounding environment
  Low-angle Dutch tilt creeping forward with dramatic floor shadows
  Anamorphic rack focus shifting from raindrops in the foreground to the emotional subject in the background

  async function generateWithGemini() {
    const apiKey = getStoredGeminiKey() || document.getElementById('geminiApiKey').value.trim();
    if (!apiKey) {
      alert('Please enter your Google Gemini API Key in the Gemini AI Engine panel above to generate with Gemini AI, or use the "⚡ Procedural Scene Slicer" which requires zero keys!');

    const style = document.getElementById('styleSelect').value;
    const influence = document.getElementById('influenceSelect').value;
    const lyrics = document.getElementById('lyricsInput').value.trim();
    const model = document.getElementById('geminiModel').value || 'gemini-3.8-flash';

    const statusEl = document.getElementById('generationStatus');
    statusEl.style.display = 'block';
    statusEl.innerText = `Connecting to Google Gemini (${model})... analyzing song structure and narrative arc...`;

    // Calculate approximate scenes
    const targetSceneCount = Math.max(3, Math.round(totalDuration / 6.0));

    const characterContext = activeCharacters.length > 0 
      ? `CAST CHARACTERS FOR VISUAL CONTINUITY:\n` + activeCharacters.map((c, i) => `${i+1}. ${c.name} (${c.role}): ${c.desc}`).join('\n')
      : `No explicit character locks; invent compelling cast fitting the aesthetic.`;

    const systemPrompt = `You are a world-class Hollywood music video director and cinematographer.
Convert the provided song lyrics, title, duration, aesthetic style, and characters into a synchronized shot-by-shot cinematic storyboard.

SONG TITLE: "${title}"
TOTAL DURATION: ${totalDuration} seconds
VISUAL AESTHETIC: ${style}
DIRECTOR INFLUENCE: ${influence}
${characterContext}

LYRICS / MUSICAL CUES:
${lyrics || '[Instrumental track with evolving emotional dynamics]'}

INSTRUCTIONS:
1. Divide the song into approximately ${targetSceneCount} consecutive scenes spanning 0 to ${totalDuration} seconds (each scene roughly 4 to 8 seconds).
2. For every scene, output:
   - sceneNumber: integer
   - startTime: string formatted MM:SS
   - endTime: string formatted MM:SS
   - duration: number of seconds
   - sectionGroup: e.g. "Intro", "Verse 1", "Chorus", "Bridge", "Outro"
   - lyricSnippet: lyric lines heard or musical feeling
   - screenplayAction: formatted narrative action and blocking
   - cameraMovement: specific cinematic camera move (dolly, crane, tracking, orbit, handheld)
   - imagePrompt: Midjourney/Flux prompt with film stock, optics, lighting, characters, and composition
   - soraPrompt: continuous motion directive for Sora/Veo with physical realism and camera movement

Return ONLY a valid JSON object matching this schema:
{
  "projectTitle": "${title}",
  "logline": "1-2 sentence executive treatment / narrative synopsis",
  "scenes": [
    {
      "sceneNumber": 1,
      "startTime": "00:00",
      "endTime": "00:06",
      "duration": 6,
      "sectionGroup": "Intro",
      "lyricSnippet": "...",
      "screenplayAction": "...",
      "cameraMovement": "...",
      "imagePrompt": "...",
      "soraPrompt": "..."
    }
  ]
}`;

    try {
      const url = `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${apiKey}`;
      const payload = {
        contents: [{ parts: [{ text: systemPrompt }] }],
        generationConfig: {
          responseMimeType: "application/json"
        }
      };

      const res = await fetch(url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });

      statusEl.innerText = `Gemini response received! Rendering storyboard and prompts...`;

  function generateScenesProcedural() {
    const title = document.getElementById('trackTitle').value || 'Cinematic Track';
    const totalDuration = Math.max(10, parseInt(document.getElementById('trackDuration').value) || 60);
    const style = document.getElementById('styleSelect').value;
    const influence = document.getElementById('influenceSelect').value;
    const rawLyrics = document.getElementById('lyricsInput').value.trim();

    const targetSceneDuration = 6.0;
    const sceneCount = Math.max(2, Math.round(totalDuration / targetSceneDuration));
    const sceneDuration = totalDuration / sceneCount;
    const lyricLines = rawLyrics ? rawLyrics.split('\n').filter(l => l.trim() && !l.trim().startsWith('[')) : [];

    const cameraMoves = [
      "Slow cinematic dolly push-in toward the subject with shallow depth of field",
      "Dynamic 360-degree orbital tracking shot keeping the subject centered",
      "Sweeping crane jib rise starting low at ground level and ascending overhead",
      "Handheld tactile follow shot with subtle organic camera breath and grain",
      "Whip pan transition tracking from the subject's gaze into the environment",
      "Low-angle Dutch tilt shot creeping forward with dramatic architectural shadows"
    ];

    const filmStocks = [
      "shot on 35mm Kodak Vision3 500T, Cooke anamorphic prime lenses, organic fine grain",
      "shot on 16mm Arri SR3, vintage high-speed glass, warm analog halation and texture",
      "shot on IMAX 70mm, Hasselblad medium format optics, razor-sharp edge-to-edge clarity",
      "shot on vintage 1970s Panavision glass with soft horizontal flares and vignette"
    ];

    const scenes = [];
    for (let i = 0; i < sceneCount; i++) {
      const startSec = i * sceneDuration;
      const endSec = (i === sceneCount - 1) ? totalDuration : (i + 1) * sceneDuration;
      const dur = Math.round((endSec - startSec) * 10) / 10;

      let groupName = 'Verse';
      const progress = (i + 0.5) / sceneCount;
      if (progress < 0.15) groupName = 'Intro';
      else if (progress > 0.85) groupName = 'Outro';
      else if (progress > 0.4 && progress < 0.65) groupName = 'Chorus';
      else if (progress >= 0.65 && progress <= 0.8) groupName = 'Bridge';

      let sceneLyric = '';
      if (lyricLines.length > 0) {
        const lineIdx = Math.floor((i / sceneCount) * lyricLines.length);
        sceneLyric = lyricLines[lineIdx] || lyricLines[lyricLines.length - 1];
      } else {
        sceneLyric = `[Instrumental ${groupName} - dynamic sonic movement]`;
      }

      const charName = activeCharacters.length > 0 ? activeCharacters[i % activeCharacters.length].name : '';
      const charDesc = activeCharacters.length > 0 ? activeCharacters[i % activeCharacters.length].desc : '';

      const cameraMove = cameraMoves[i % cameraMoves.length];
      const stock = filmStocks[i % filmStocks.length];

      const imagePrompt = `${charName ? `${charName} (${charDesc}). ` : ''}${sceneLyric}. ${style}. Director influence: ${influence}. Cinematic framing, ${stock}, dramatic high contrast studio lighting, 8k resolution.`;

      const soraPrompt = `A continuous ${dur}s cinematic shot in ${style} aesthetic, influenced by ${influence}. ${charName ? `${charName} moves with authentic weight and gaze. ` : ''}${sceneLyric}. ${cameraMove}. Authentic micro-gestures and physical lighting.`;

      scenes.push({
        sceneNumber: i + 1,
        startTime: formatSeconds(startSec),
        endTime: formatSeconds(endSec),
        duration: dur,
        sectionGroup: groupName,
        lyricSnippet: sceneLyric,
        screenplayAction: `INT./EXT. SCENE ${i+1} - ${groupName.toUpperCase()}\n${charName ? `${charName.toUpperCase()} is captured on screen. ` : ''}Camera performs ${cameraMove.toLowerCase()} while musical elements resonate.`,
        cameraMovement: cameraMove,
        imagePrompt: imagePrompt,
        soraPrompt: soraPrompt
      });
    }

    currentGeneratedScenes = scenes;
    renderStoryboard(scenes, `Cinematic narrative exploring "${title}" through a ${style} aesthetic.`);
  }

      }

      // Card Background
      doc.setFillColor(250, 250, 250);
      doc.setDrawColor(220, 225, 230);
      doc.roundedRect(margin, cursorY, contentWidth, cardHeight, 6, 6, 'FD');

      // Card Header
      doc.setFont('helvetica', 'bold');
      doc.setFontSize(10);
      doc.setTextColor(37, 99, 235);
      doc.text(`SCENE ${s.sceneNumber}  [${s.startTime} - ${s.endTime} • ${s.duration}s]`, margin + 12, cursorY + 18);

      doc.setFont('helvetica', 'bold');
      doc.setTextColor(100, 116, 139);
      doc.text(s.sectionGroup ? s.sectionGroup.toUpperCase() : 'SCENE', pageWidth - margin - 80, cursorY + 18);

      doc.setFont('helvetica', 'italic');
      doc.setFontSize(9);
      doc.setTextColor(180, 83, 9);
      doc.text(`Lyric: "${s.lyricSnippet}"`, margin + 12, cursorY + 32);

      let textY = cursorY + 48;

      // Image Prompt
      doc.setFont('helvetica', 'normal');
      doc.setFontSize(8.5);
      doc.setTextColor(30, 41, 59);
      doc.text(promptLines, margin + 12, textY);
      textY += (promptLines.length * 11) + 4;

      // Sora Prompt
      doc.setTextColor(5, 150, 105);
      doc.text(soraLines, margin + 12, textY);

      cursorY += cardHeight + 12;
    }

    // Add page numbers
    const totalPages = doc.internal.getNumberOfPages();
    for (let p = 1; p <= totalPages; p++) {
      doc.setPage(p);
      addHeaderAndFooter(p, totalPages);
    }

    const cleanName = title.toLowerCase().replace(/[^a-z0-9]/g, '_').replace(/_+/g, '_');
    doc.save(`${cleanName}_master_production_notes.pdf`);
  }

  function copyAllPrompts() {
    const cards = document.querySelectorAll('.scene-card');
    if (!cards || cards.length === 0) {
      alert('Generate scenes first before copying!');
      return;
    }

    let text = '=== AUDIOARC SCENE PROMPTS ===\n\n';
    cards.forEach((card, idx) => {
      const title = card.querySelector('.scene-title')?.innerText || `Scene ${idx+1}`;
      const time = card.querySelector('.scene-badge-time')?.innerText || '';
      const lyric = card.querySelector('.scene-lyric')?.innerText || '';
      const boxes = card.querySelectorAll('.prompt-box');

      text += `${title} ${time}\nLyric: ${lyric}\n`;
      boxes.forEach(b => {
        text += b.innerText + '\n';
      });
      text += '\n----------------------------------------\n\n';
    });

    navigator.clipboard.writeText(text).then(() => {
      alert('Copied all scene prompts to clipboard!');
    });
  }
